// ==UserScript==
// @name         YouTube Android – traducere subtitrări în română
// @namespace    local.yt.mobile.ro
// @version      1.1
// @description  Pentru m.youtube.com: subtitrări → Traducere automată → Română. O singură încercare per videoclip.
// @match        https://m.youtube.com/*
// @grant        none
// @run-at       document-idle
// ==/UserScript==

(() => {
  'use strict';

  const WAIT_FOR_MENU_MS = 7000;
  const POLL_MS = 250;
  const attemptedVideos = new Set();
  let running = false;

  const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

  function videoId() {
    return new URL(location.href).searchParams.get('v') || location.pathname;
  }

  function visible(el) {
    if (!el) return false;
    const rect = el.getBoundingClientRect();
    const style = getComputedStyle(el);
    return rect.width > 0 && rect.height > 0 &&
      style.visibility !== 'hidden' &&
      style.display !== 'none';
  }

  function label(el) {
    return [
      el.innerText,
      el.getAttribute('aria-label'),
      el.getAttribute('title')
    ].filter(Boolean).join(' ').trim().toLowerCase();
  }

  function clickOnce(el) {
    if (!el || !visible(el)) return false;
    el.click();
    return true;
  }

  function findButton(matches) {
    const buttons = [...document.querySelectorAll(
      'button, [role="button"], .ytp-button'
    )].filter(visible);

    return buttons.find(el => matches.some(test => test(label(el))) ) || null;
  }

  function findMenuItem(matches) {
    const items = [...document.querySelectorAll(
      '.ytp-menuitem, [role="menuitem"], [role="option"], yt-list-item-view-model'
    )].filter(visible);

    return items.find(el => matches.some(test => test(label(el)))) || null;
  }

  async function waitFor(find, timeout = WAIT_FOR_MENU_MS) {
    const end = Date.now() + timeout;
    while (Date.now() < end) {
      const result = find();
      if (result) return result;
      await sleep(POLL_MS);
    }
    return null;
  }

  function scrollableAncestors(el) {
    const result = [];
    let node = el?.parentElement;

    while (node && node !== document.body) {
      const style = getComputedStyle(node);
      if (/(auto|scroll)/.test(style.overflowY) &&
          node.scrollHeight > node.clientHeight) {
        result.push(node);
      }
      node = node.parentElement;
    }

    return result;
  }

  async function findAndScrollToRomanian() {
    const isRomanian = el => {
      const text = label(el);
      return text === 'română' || text === 'romana' ||
        text === 'romanian' || text.includes('română') ||
        text.includes('romana') || text.includes('romanian');
    };

    const end = Date.now() + WAIT_FOR_MENU_MS;
    let steps = 0;

    while (Date.now() < end && steps < 24) {
      const candidates = [...document.querySelectorAll(
        'yt-list-item-view-model, [role="menuitem"], [role="option"], button'
      )];

      const romanian = candidates.find(isRomanian);
      if (romanian && visible(romanian)) {
        romanian.scrollIntoView({ block: 'center', behavior: 'smooth' });
        await sleep(250);
        return clickOnce(romanian);
      }

      // Derulează panoul de limbi, dacă există; altfel încearcă derularea paginii.
      const panelItem = document.querySelector(
        'yt-list-item-view-model.ytListItemViewModelHost:nth-of-type(10)'
      );
      const scrollTargets = scrollableAncestors(panelItem);

      if (scrollTargets.length) {
        for (const target of scrollTargets) {
          target.scrollTop += Math.max(250, Math.floor(target.clientHeight * 0.7));
        }
      } else {
        window.scrollBy(0, Math.max(250, Math.floor(window.innerHeight * 0.7)));
      }

      steps++;
      await sleep(POLL_MS);
    }

    return false;
  }

  async function runForVideo(id) {
    if (running || attemptedVideos.has(id)) return;
    running = true;
    attemptedVideos.add(id);

    try {
      await sleep(1800);
      if (id !== videoId()) return;

      const cc = findButton([
        t => t.includes('captions') || t.includes('subtitles') ||
             t.includes('subtitr')
      ]);

      if (cc) {
        const pressed = cc.getAttribute('aria-pressed');
        const isOn = pressed === 'true' ||
          /\b(on|activat|pornit)\b/i.test(label(cc));

        if (!isOn) clickOnce(cc);
        await sleep(500);
      }

      if (id !== videoId()) return;

      const settings = findButton([
        t => t.includes('settings') || t.includes('setări') ||
             t.includes('setari')
      ]);
      if (!clickOnce(settings)) return;

      const subtitles = await waitFor(() => findMenuItem([
        t => t.includes('subtitles') || t.includes('subtitr') ||
             t.includes('captions')
      ]));
      if (!clickOnce(subtitles)) return;

      const autoTranslate = await waitFor(() => {
        // Selectorul indicat, cu potrivirea după text ca rezervă.
        const bySelector = document.querySelector(
          'yt-list-item-view-model.ytListItemViewModelHost:nth-of-type(10)'
        );

        if (visible(bySelector) &&
            /auto-translate|translate automatically|traducere automată|traduce automat/i
              .test(label(bySelector))) {
          return bySelector;
        }

        return findMenuItem([
          t => t.includes('auto-translate') ||
               t.includes('translate automatically') ||
               t.includes('traducere automată') ||
               t.includes('traduce automat')
        ]);
      });

      if (!clickOnce(autoTranslate)) return;

      // Așteaptă panoul de limbi, apoi îl derulează până la Română.
      await sleep(350);
      await findAndScrollToRomanian();
    } finally {
      running = false;
    }
  }

  function checkVideo() {
    const id = videoId();
    if (id && !attemptedVideos.has(id)) {
      runForVideo(id);
    }
  }

  const originalPushState = history.pushState;
  history.pushState = function (...args) {
    const result = originalPushState.apply(this, args);
    setTimeout(checkVideo, 1200);
    return result;
  };

  const originalReplaceState = history.replaceState;
  history.replaceState = function (...args) {
    const result = originalReplaceState.apply(this, args);
    setTimeout(checkVideo, 1200);
    return result;
  };

  window.addEventListener('popstate', () => setTimeout(checkVideo, 1200));
  setTimeout(checkVideo, 1800);
})();
